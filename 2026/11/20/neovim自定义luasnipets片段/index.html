local ls = require("luasnip")
local s  = ls.snippet
local t  = ls.text_node
local i  = ls.insert_node
local c  = ls.choice_node
local f  = ls.function_node

local date = function() return os.date("%Y-%m-%d") end
local time = function() return os.date("%H:%M") end

-- 生成对称 # 的标题：h = 层级
local function heading(level, default)
  local h = string.rep("#", level)
  return {
    t(h .. " "),        -- 前置 #
    i(1, default),      -- 标题文字
    t(" " .. h),        -- 后置 #，数量与前置一致
    t({ "", "" }),      -- 空行
    i(0),
  }
end

return {
  -- ===== 标题（沿用原 sec 系列，习惯不变）=====
  s("sec",   heading(1, "Section Name")),
  s("ssec",  heading(2, "Section Name")),
  s("sssec", heading(3, "Section Name")),
  s("par",   heading(4, "Paragraph Name")),
  s("spar",  heading(5, "Paragraph Name")),

  -- ===== 链接 / 图片 =====
  s("link",  { t("["), i(1, "text"), t("]("), i(2, "url"), t(")"), i(0) }),
  s("img",   { t("!["), i(1, "alt"), t("]("), i(2, "url"), t(")"), i(0) }),
  s("imgw",  { t("!["), i(1, "alt"), t("]("), i(2, "url"), t(" ="), i(3, "500x"), t(")"), i(0) }),
  s("rlink", { t("["), i(1, "text"), t("]["), i(2, "1"), t("]"), i(0) }),

  -- ===== 行内格式 =====
  s("bold",  { t("**"), i(1), t("**"), i(0) }),
  s("ital",  { t("*"),  i(1), t("*"),  i(0) }),
  s("bi",    { t("***"), i(1), t("***"), i(0) }),
  s("del",   { t("~~"), i(1), t("~~"), i(0) }),
  s("mark",  { t("=="), i(1), t("=="), i(0) }),
  s("code",  { t("`"),  i(1), t("`"),  i(0) }),
  s("cmt",   { t("<!-- "), i(1), t(" -->"), i(0) }),

  -- ===== 块级 =====
  s("codeblock", { t("```"), i(1, "lua"), t({ "", "" }), i(2), t({ "", "```", "" }), i(0) }),
  s("quote",  { t("> "), i(1), t({ "", "" }), i(0) }),
  s("todo",   { t("- [ ] "), i(1), t({ "", "" }), i(0) }),
  s("hr",     { t({ "---", "" }), i(0) }),

  s("table", {
    t("| "), i(1, "Header"), t(" | "), i(2, "Header"), t({ " |", "| --- | --- |", "| " }),
    i(3), t(" | "), i(4), t({ " |", "" }), i(0),
  }),

  -- ===== frontmatter：自动填日期 =====
  s("fm", {
    t({ "---", "title: " }), i(1, "Title"),
    t({ "date: " }), f(date),
    t({ "tags: []", "---", "" }), i(0),
  }),

  -- ===== 脚注 / 目录 =====
  s("fn",  { t("[^"), i(1, "1"), t("]"), t({ "", "" }), i(0) }),
  s("toc", { t({ "<!-- toc -->", "" }), i(0) }),

  -- ===== 折叠 / Callout =====
  s("det", {
    t("<details>"), t({ "", "<summary>" }), i(1, "Summary"),
    t({ "", "</summary>", "" }), i(2),
    t({ "", "</details>", "" }), i(0),
  }),

  s("note",   { t("> [!NOTE]"),   t({ "", "> " }), i(1), t({ "", "" }), i(0) }),
  s("tip",    { t("> [!TIP]"),    t({ "", "> " }), i(1), t({ "", "" }), i(0) }),
  s("warn",   { t("> [!WARNING]"), t({ "", "> " }), i(1), t({ "", "" }), i(0) }),
  s("danger", { t("> [!CAUTION]"), t({ "", "> " }), i(1), t({ "", "" }), i(0) }),

  -- ===== 数学 / mermaid =====
  s("math",  { t("$"), i(1), t("$"), i(0) }),
  s("dmath", { t({ "$$", "" }), i(1), t({ "", "$$", "" }), i(0) }),
  s("mmd", {
    t({ "```mermaid", "graph TD" }), t({ "", "  A[Start] --> B[End]" }),
    t({ "", "```", "" }), i(0),
  }),

  -- ===== 时间 =====
  s("date", { f(date), i(0) }),
  s("time", { f(time), i(0) }),

  -- ===== Obsidian =====
  s("wl",  { t("[["), i(1, "note"), t("]]"), i(0) }),
  s("blk", { t(" ^"), i(1, "blockid"), i(0) }),
}

return function()
  local ls = require("luasnip")
  ---------------------------------------------------------------
  -- 1) 基础设置
  ---------------------------------------------------------------
  ls.setup({
    history = true,
    updateevents = "TextChanged,TextChangedI",
    enable_autosnippets = true,
  })

  ---------------------------------------------------------------
  -- 2) 加载 snippet 目录（路径信息在这里）
  ---------------------------------------------------------------
  require("luasnip.loaders.from_vscode").load()
  require("luasnip.loaders.from_vscode").load({
    paths = vim.fn.expand("~/.vsnip"),
  })
  require("luasnip.loaders.from_lua").load({
    paths = vim.fn.expand("~/.luasnip"),
  })

  ---------------------------------------------------------------
  -- 3) choiceNode 切换键
  ---------------------------------------------------------------
  -- 菜单选择
  vim.keymap.set({ "i", "s" }, "<C-e>", function()
      if require("luasnip").choice_active() then
          require("luasnip.extras.select_choice")()
      end
  end, { silent = true })

  vim.keymap.set({ "i", "s" }, "<C-l>", function()
    if ls.choice_active() then ls.change_choice(1) end
  end, { silent = true, desc = "LuaSnip: next choice" })

  vim.keymap.set({ "i", "s" }, "<C-h>", function()
    if ls.choice_active() then ls.change_choice(-1) end
  end, { silent = true, desc = "LuaSnip: previous choice" })

  ---------------------------------------------------------------
  -- 4) 跳转键
  ---------------------------------------------------------------
  vim.keymap.set({"i", "s"}, "<Tab>", function()
     -- 优先展开或向前跳转
     if ls.expand_or_jumpable() then
	ls.expand_or_jump()
     end
  end, { silent = true, desc = "LuaSnip: expand or jump" })

  vim.keymap.set({"i", "s"}, "<S-Tab>", function()
     -- 仅在确实可以向左跳转时，才执行向后跳转
     if ls.jumpable(-1) then
	ls.jump(-1)
     end
  end, { silent = true, desc = "LuaSnip: jump backward" })
end
注意

最初发现使用<Tab>并不能跳转，比如在 LaTeX 文件中输入 pdv, 则自动展开为 \pdv{}{}, 同时光标位于第一个花括号内，但是按<Tab>时其并不能跳转到第二个花括号内，使用命令:verbose imap <Tab> 查询到其被 Ultisnips占用，于是删除这个老旧的插件后恢复正常。最后该文件需要在~/.config/nvim/init.lua中使用require('plg/luasnip')调用才行。
